FUNCTION_to_html_css():

    return _to_html(lines, source_name, source_path, css_rel_path)

FUNCTIONconvert_project():

    set project_dir to Path(project_path).resolve()

    if not project_dir.is_dir():

        raise ValueError(f"Not a directory: {project_dir}")

    set output_base to MIRRORS_DIR / project_dir.name

    set py_files to sorted(project_dir.rglob("*.py"))

    if not py_files:

        print(f"No .py files found in {project_dir}")

        return

    print(f"Converting {len(py_files)} file(s) from: {project_dir}")

    print(f"Output root: {output_base}\n")

    for each src_path in py_files:

        set rel_path to src_path.relative_to(project_dir)

        set output_path to output_base / rel_path.with_suffix(".html")

        set depth to len(output_path.relative_to(MIRRORS_DIR).parts) - 1

        set css_rel_path to ("../" * depth) + "pseudo.css"

        set src_bytes to src_path.read_bytes()

        set tree to _parse(src_bytes)

        set lines to []

        _convert(tree.root_node, 0, lines)

        output_path.parent.mkdir(parents=True, exist_ok=True)

        output_path.write_text( _to_html_css(lines, src_path.name, str(project_dir / rel_path), css_rel_path), encoding="utf-8", )

        print(f" {rel_path} → {output_path.relative_to(MIRRORS_DIR.parent)}")

    print(f"\nDone.")

if __name__ == "__main__":

    if len(sys.argv) < 2:

        print("Usage: python project_to_pseudo.py <path/to/project_dir>")

        sys.exit(1)

    convert_project(sys.argv[1])
