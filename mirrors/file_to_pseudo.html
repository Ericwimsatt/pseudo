set PY_LANGUAGE to Language(tspython.language())

set MIRRORS_DIR to Path(__file__).parent / "mirrors"

FUNCTION_make_parser():

    return Parser(PY_LANGUAGE)

FUNCTION_parse():

    return _make_parser().parse(src_bytes)

FUNCTION_text(node):

    return node.text.decode("utf-8", errors="replace")

FUNCTION_node_line(node):

    return node.start_point[0] + 1

// ---------------------------------------------------------------------------

// Line-level transformers — each returns a list of (text_or_html, is_html)

// ---------------------------------------------------------------------------

FUNCTION_for_header(node):

    set left to node.child_by_field_name("left")

    set right to node.child_by_field_name("right")

    set left_text to _text(left) if left else "?"

    set right_text to _text(right) if right else "?"

    if left_text == "_" and right and right.type == "call":

        set fn to right.child_by_field_name("function")

        set args to right.child_by_field_name("arguments")

        if fn and _text(fn) == "range" and args:

            set named to args.named_children

            if len(named) == 1:

                return [("do", False), (f" {_text(named[0])} times", False)]

    if right and right.type == "call":

        set fn to right.child_by_field_name("function")

        set args to right.child_by_field_name("arguments")

        if fn and _text(fn) == "range" and args:

            set named to args.named_children

            if len(named) == 1:

                set n to _text(named[0])

                return [("for", False), (f" {left_text} from 0 to {n} - 1", False)]

    return [("for each", False), (f" {left_text} in {right_text}", False)]

FUNCTION_function_header(node):

    set name to node.child_by_field_name("name")

    set params to node.child_by_field_name("parameters")

    set name_text to _text(name) if name else "?"

    set name_line to _node_line(name) if name else 0

    set param_parts to []

    set param_names to []

    if params:

        for each child in params.named_children:

            set t to child.type

            if t == "identifier":

                set pn to _text(child)

                param_parts.append(pn)

                param_names.append(pn)

            else if t in ("default_parameter", "typed_parameter", "typed_default_parameter"):

                set n to child.child_by_field_name("name")

                if n:

                    set pn to _text(n)

                    param_parts.append(pn)

                    param_names.append(pn)

            else if t == "list_splat_pattern":

                set inner to child.named_children[0] if child.named_children else None

                param_parts.append("*" + (_text(inner) if inner else "args"))

            else if t == "dictionary_splat_pattern":

                set inner to child.named_children[0] if child.named_children else None

                param_parts.append("**" + (_text(inner) if inner else "kwargs"))

    set parts to [('<span class="kw">FUNCTION</span>', True)]

    set attrs to f' class="fn-name" data-fn="{name_text}" data-line="{name_line}"'

    parts.append((f'<span{attrs}>{name_text}</span>', True))

    parts.append((f'({", ".join(param_parts)})', False))

    return parts

FUNCTION_assignment_line(node):

    set left to node.child_by_field_name("left")

    set right to node.child_by_field_name("right")

    set line to _node_line(node)

    set left_text to _text(left) if left else "?"

    set right_text to _text(right) if right else "?"

    set parts to [("set ", False)]

    if left and left.type == "identifier":

        set vname to _text(left)

        set line2 to _node_line(left)

        set attrs to f' class="var-name" data-var="{vname}" data-line="{line2}"'

        parts.append((f'<span{attrs}>{left_text}</span>', True))

    otherwise:

        parts.append((left_text, False))

    parts.append((f" to {right_text}", False))

    return parts

FUNCTION_var_span(left_node):

    set line2 to _node_line(left_node)

    set attrs to f' class="var-name" data-var="{vname}" data-line="{line2}"'

    return (f'<span{attrs}>{vname}</span>', True)

FUNCTION_augmented_assignment_line(node):

    set left to node.child_by_field_name("left")

    set right to node.child_by_field_name("right")

    set op_node to node.child_by_field_name("operator")

    set left_text to _text(left) if left else "?"

    set right_text to _text(right) if right else "?"

    if op_node is None:

        for each child in node.children:

            if child.type in ("+=", "-=", "*=", "/=", "//=", "%=", "**=", "&=", "|=", "^="):

                set op_node to child

                break

    set op to _text(op_node) if op_node else "?"

    set left_seg to _var_span(left, left_text) if (left and left.type == "identifier") else (left_text, False)

    set right_seg to _var_span(right, right_text) if (right and right.type == "identifier") else (right_text, False)

    set ops to { "+=": lambda: [("add ", False), right_seg, (" to ", False), left_seg], "-=": lambda: [("subtract ", False), right_seg, (" from ", False), left_seg], "*=": lambda: [("multiply ", False), left_seg, (" by ", False), right_seg], "/=": lambda: [("divide ", False), left_seg, (" by ", False), right_seg], "//=": lambda: [("floor-divide ", False), left_seg, (" by ", False), right_seg], "%=": lambda: [left_seg, (" mod= ", False), right_seg], "**=": lambda: [("raise ", False), left_seg, (" to the power of ", False), right_seg], }

    if op in ops:

        return ops[op]()

    return [(f"{left_text} {op} {right_text}", False)]

FUNCTION_return_line(node):

    set named to node.named_children

    if named:

        return [(f"return {_text(named[0])}", False)]

    return [("return", False)]

// Main recursive converter — appends (depth, parts, css_class) tuples

// parts is a list of (text_or_html, is_html_bool)

// ---------------------------------------------------------------------------

FUNCTION_convert(node):

    set t to node.type

    if t == "module":

        for each child in node.named_children:

            _convert(child, depth, out)

    else if t == "block":

        for each child in node.named_children:

            _convert(child, depth, out)

    else if t == "comment":

        set text to _text(node)[1:].strip()

        out.append((depth, [(f"// {text}", False)], "pseudo-comment"))

    else if t in ("import_statement", "import_from_statement"):

        pass

    else if t == "function_definition":

        set parts to _function_header(node)

        parts.append((":", False))

        out.append((depth, parts, "pseudo-function"))

        set body to node.child_by_field_name("body")

        if body:

            _convert(body, depth + 1, out)

    else if t == "if_statement":

        set cond to node.child_by_field_name("condition")

        out.append((depth, [(f"if {_text(cond) if cond else '?'}:", False)], "pseudo-if"))

        set consequence to node.child_by_field_name("consequence")

        if consequence:

            _convert(consequence, depth + 1, out)

        for each child in node.named_children:

            if child.type == "elif_clause":

                set cond2 to child.child_by_field_name("condition")

                out.append((depth, [(f"else if {_text(cond2) if cond2 else '?'}:", False)], "pseudo-if"))

                set cons2 to child.child_by_field_name("consequence")

                if cons2:

                    _convert(cons2, depth + 1, out)

            else if child.type == "else_clause":

                out.append((depth, [("otherwise:", False)], "pseudo-if"))

                set body to child.child_by_field_name("body")

                if body:

                    _convert(body, depth + 1, out)

    else if t == "for_statement":

        set parts to _for_header(node)

        parts.append((":", False))

        out.append((depth, parts, "pseudo-for"))

        set body to node.child_by_field_name("body")

        if body:

            _convert(body, depth + 1, out)

    else if t == "while_statement":

        set cond to node.child_by_field_name("condition")

        out.append((depth, [(f"while {_text(cond) if cond else '?'}:", False)], "pseudo-while"))

        set body to node.child_by_field_name("body")

        if body:

            _convert(body, depth + 1, out)

    else if t == "assignment":

        out.append((depth, _assignment_line(node), "pseudo-assignment"))

    else if t == "augmented_assignment":

        out.append((depth, _augmented_assignment_line(node), "pseudo-assignment"))

    else if t == "return_statement":

        out.append((depth, _return_line(node), "pseudo-return"))

    else if t == "expression_statement":

        for each child in node.named_children:

            if child.type in ("string", "concatenated_string"):

                pass

            otherwise:

                _convert(child, depth, out)

    else if t == "pass_statement":

        out.append((depth, [("pass", False)], "pseudo-control"))

    else if t == "break_statement":

        out.append((depth, [("break", False)], "pseudo-control"))

    else if t == "continue_statement":

        out.append((depth, [("continue", False)], "pseudo-control"))

    otherwise:

        set raw to _text(node).strip()

        if raw:

            out.append((depth, [(raw, False)], "pseudo-raw"))

FUNCTION_escape():

    return text.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")

FUNCTION_to_html(css_rel_path):

    set parts to [ "<!DOCTYPE html>", "<html>", "<head>", f"<title>Pseudocode: {source_name}</title>", f'<link rel="stylesheet" href="{_escape(css_rel_path)}">', _HTMX_JS, "</head>", "<body>", ]

    for each depth, segs, css_class in lines:

        set indent to "&nbsp;&nbsp;&nbsp;&nbsp;" * depth

        set inner to ""

        for each text_or_html, is_html in segs:

            if is_html:

                add text_or_html to inner

            otherwise:

                add _escape(text_or_html) to inner

        set attrs to f' class="{css_class}" data-source-file="{_escape(file_path)}"'

        parts.append(f"<p{attrs}>{indent}{inner}</p>")

    parts.append(_MODAL_HTML)

    parts.append(_INTERACTIVE_JS)

    add ["</body>", "</html>"] to parts

    return "\n".join(parts)

FUNCTIONconvert_file(output_path):

    set src_path to Path(input_path).resolve()

    set src_bytes to src_path.read_bytes()

    set tree to _parse(src_bytes)

    set lines to []

    _convert(tree.root_node, 0, lines)

    MIRRORS_DIR.mkdir(exist_ok=True)

    if output_path is None:

        set output_path to MIRRORS_DIR / (src_path.stem + ".html")

    otherwise:

        set output_path to Path(output_path)

    output_path.write_text( _to_html(lines, src_path.name, str(src_path)), encoding="utf-8", )

    return output_path
