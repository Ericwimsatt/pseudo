set PY_LANGUAGE to Language(tspython.language())

set MIRRORS_DIR to Path(__file__).parent / "mirrors"

// ---------------------------------------------------------------------------

// Parser

// ---------------------------------------------------------------------------

function _make_parser():

    return Parser(PY_LANGUAGE)

function _parse():

    return _make_parser().parse(src_bytes)

// ---------------------------------------------------------------------------

// Node helpers

// ---------------------------------------------------------------------------

function _text(node):

    return node.text.decode("utf-8", errors="replace")

// ---------------------------------------------------------------------------

// Statement-level transformers

// Each returns a plain-text pseudocode string (no HTML).

// ---------------------------------------------------------------------------

function _for_header(node):

    set left to node.child_by_field_name("left")

    set right to node.child_by_field_name("right")

    set left_text to _text(left) if left else "?"

    set right_text to _text(right) if right else "?"

    // for _ in range(N)

    if left_text == "_" and right and right.type == "call":

        set fn to right.child_by_field_name("function")

        set args to right.child_by_field_name("arguments")

        if fn and _text(fn) == "range" and args:

            set named to args.named_children

            if len(named) == 1:

                return f"do {_text(named[0])} times"

    // for x in range(N)

    if right and right.type == "call":

        set fn to right.child_by_field_name("function")

        set args to right.child_by_field_name("arguments")

        if fn and _text(fn) == "range" and args:

            set named to args.named_children

            if len(named) == 1:

                set n to _text(named[0])

                return f"for {left_text} from 0 to {n} - 1"

    // default

    return f"for each {left_text} in {right_text}"

function _function_header(node):

    set name to node.child_by_field_name("name")

    set params to node.child_by_field_name("parameters")

    set name_text to _text(name) if name else "?"

    set param_parts to []

    if params:

        for each child in params.named_children:

            set t to child.type

            if t == "identifier":

                param_parts.append(_text(child))

            else if t in ("default_parameter", "typed_parameter", "typed_default_parameter"):

                set n to child.child_by_field_name("name")

                if n:

                    param_parts.append(_text(n))

            else if t == "list_splat_pattern":

                set inner to child.named_children[0] if child.named_children else None

                param_parts.append("*" + (_text(inner) if inner else "args"))

            else if t == "dictionary_splat_pattern":

                set inner to child.named_children[0] if child.named_children else None

                param_parts.append("**" + (_text(inner) if inner else "kwargs"))

    return f"function {name_text}({', '.join(param_parts)})"

function _assignment_line(node):

    set left to node.child_by_field_name("left")

    set right to node.child_by_field_name("right")

    set left_text to _text(left) if left else "?"

    set right_text to _text(right) if right else "?"

    return f"set {left_text} to {right_text}"

function _augmented_assignment_line(node):

    set left to node.child_by_field_name("left")

    set right to node.child_by_field_name("right")

    set op_node to node.child_by_field_name("operator")

    set left_text to _text(left) if left else "?"

    set right_text to _text(right) if right else "?"

    // Fallback: search anonymous children for the operator token

    if op_node is None:

        for each child in node.children:

            if child.type in ("+=", "-=", "*=", "/=", "//=", "%=", "**=", "&=", "|=", "^="):

                set op_node to child

                break

    set op to _text(op_node) if op_node else "?"

    set ops to { "+=": f"add {right_text} to {left_text}", "-=": f"subtract {right_text} from {left_text}", "*=": f"multiply {left_text} by {right_text}", "/=": f"divide {left_text} by {right_text}", "//=": f"floor-divide {left_text} by {right_text}", "%=": f"{left_text} mod= {right_text}", "**=": f"raise {left_text} to the power of {right_text}", }

    return ops.get(op, f"{left_text} {op} {right_text}")

function _return_line(node):

    set named to node.named_children

    if named:

        return f"return {_text(named[0])}"

    return "return"

// ---------------------------------------------------------------------------

// Main recursive converter

// Appends (depth: int, text: str) tuples to `out`.

// ---------------------------------------------------------------------------

function _convert(node):

    set t to node.type

    if t == "module":

        for each child in node.named_children:

            _convert(child, depth, out)

    else if t == "block":

        for each child in node.named_children:

            _convert(child, depth, out)

    else if t == "comment":

        set text to _text(node)[1:].strip()

        out.append((depth, f"// {text}", "pseudo-comment"))

    else if t in ("import_statement", "import_from_statement"):

        pass

        // skipped per transformations table

    else if t == "function_definition":

        out.append((depth, _function_header(node) + ":", "pseudo-function"))

        set body to node.child_by_field_name("body")

        if body:

            _convert(body, depth + 1, out)

    else if t == "if_statement":

        set cond to node.child_by_field_name("condition")

        out.append((depth, f"if {_text(cond) if cond else '?'}:", "pseudo-if"))

        set consequence to node.child_by_field_name("consequence")

        if consequence:

            _convert(consequence, depth + 1, out)

        // elif and else are named_children of the if_statement

        for each child in node.named_children:

            if child.type == "elif_clause":

                set cond2 to child.child_by_field_name("condition")

                out.append((depth, f"else if {_text(cond2) if cond2 else '?'}:", "pseudo-if"))

                set cons2 to child.child_by_field_name("consequence")

                if cons2:

                    _convert(cons2, depth + 1, out)

            else if child.type == "else_clause":

                out.append((depth, "otherwise:", "pseudo-if"))

                set body to child.child_by_field_name("body")

                if body:

                    _convert(body, depth + 1, out)

    else if t == "for_statement":

        out.append((depth, _for_header(node) + ":", "pseudo-for"))

        set body to node.child_by_field_name("body")

        if body:

            _convert(body, depth + 1, out)

    else if t == "while_statement":

        set cond to node.child_by_field_name("condition")

        out.append((depth, f"while {_text(cond) if cond else '?'}:", "pseudo-while"))

        set body to node.child_by_field_name("body")

        if body:

            _convert(body, depth + 1, out)

    else if t == "assignment":

        out.append((depth, _assignment_line(node), "pseudo-assignment"))

    else if t == "augmented_assignment":

        out.append((depth, _augmented_assignment_line(node), "pseudo-assignment"))

    else if t == "return_statement":

        out.append((depth, _return_line(node), "pseudo-return"))

    else if t == "expression_statement":

        for each child in node.named_children:

            if child.type in ("string", "concatenated_string"):

                pass

                // skip docstrings

            otherwise:

                _convert(child, depth, out)

    else if t == "pass_statement":

        out.append((depth, "pass", "pseudo-control"))

    else if t == "break_statement":

        out.append((depth, "break", "pseudo-control"))

    else if t == "continue_statement":

        out.append((depth, "continue", "pseudo-control"))

    otherwise:

        set raw to _text(node).strip()

        if raw:

            out.append((depth, raw, "pseudo-raw"))

// ---------------------------------------------------------------------------

// HTML writer

// ---------------------------------------------------------------------------

function _escape():

    return text.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")

function _to_html():

    set parts to [ "<!DOCTYPE html>", "<html>", "<head>", f"<title>Pseudocode: {source_name}</title>", '<link rel="stylesheet" href="pseudo.css">', "</head>", "<body>", ]

    for each depth, text, css_class in lines:

        set indent to "&nbsp;&nbsp;&nbsp;&nbsp;" * depth

        parts.append(f'<p class="{css_class}">{indent}{_escape(text)}</p>')

    add ["</body>", "</html>"] to parts

    return "\n".join(parts)

// ---------------------------------------------------------------------------

// Public API

// ---------------------------------------------------------------------------

function convert_file():

    set src_path to Path(input_path).resolve()

    set src_bytes to src_path.read_bytes()

    set tree to _parse(src_bytes)

    set lines to []

    _convert(tree.root_node, 0, lines)

    MIRRORS_DIR.mkdir(exist_ok=True)

    set output_path to MIRRORS_DIR / (src_path.stem + ".html")

    output_path.write_text(_to_html(lines, src_path.name), encoding="utf-8")

    return output_path

if __name__ == "__main__":

    if len(sys.argv) < 2:

        print("Usage: python file_to_pseudo.py <path/to/file.py>")

        sys.exit(1)

    set result to convert_file(sys.argv[1])

    print(f"Written to: {result}")
