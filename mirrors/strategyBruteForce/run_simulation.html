FUNCTIONcalc_blind_payout(player_rank):

    if player_rank <= 1:

        return 500

    else if player_rank <= 10:

        return 50

    else if player_rank <= 166:

        return 10

    else if player_rank <= 322:

        return 3

    else if player_rank <= 1599:

        return 1.5

    else if player_rank <= 1609:

        return 1

    otherwise:

        return 0

FUNCTIONplay_game(pre_flop_strategy, post_flop_strategy, river_strategy, dead_card_maker):

    set deck to Deck()

    set player_hand to sorted(deck.draw(2))

    set board to []

    set dead_cards to deck.pull_many(dead_card_maker(player_hand))

    set dealer_hand to deck.draw(2)

    set play_bet to 0

    if pre_flop_strategy(player_hand, dead_cards):

        set play_bet to 4

    add deck.draw(3) to board

    if play_bet == 0 and post_flop_strategy(player_hand, board, dead_cards):

        set play_bet to 2

    add deck.draw(2) to board

    if play_bet == 0:

        if river_strategy(player_hand, board, dead_cards):

            set play_bet to 1

    // print("Evaling hand")

    // Card.print_pretty_cards(player_hand)

    // Card.print_pretty_cards(board)

    // Card.print_pretty_cards(dealer_hand)

    // evaluation.hand_summary(board, [player_hand, dealer_hand])

    set dealer_rank to evaluation.evaluate(dealer_hand, board)

    set player_rank to evaluation.evaluate(player_hand, board)

    set ante_bet to 0

    // Check if dealer qualifies

    // 6185 is max rank for a pair

    if dealer_rank <= 6185:

        set ante_bet to 1

    set result to 0

    // fold

    if play_bet == 0:

        set result to -2

    else if player_rank == dealer_rank:

        set result to 0

    else if player_rank < dealer_rank:

        set blind_payout to calc_blind_payout(player_rank)

        set result to play_bet + blind_payout + ante_bet

    otherwise:

        set result to - play_bet - ante_bet - 1

    return result, { "player_wins": 1 if result > 0 else 0, "dealer_wins": 1 if result < 0 else 0, "pushes": 1 if result == 0 else 0, "dealer_qualifies": 1 if ante_bet else 0, "play_bet_total": play_bet, "player_win_play_bet_total": play_bet if result > 0 else 0, "dealer_win_play_bet_total": play_bet if result < 0 else 0, "push_play_bet_total": play_bet if result == 0 else 0, "blind_payout_total": blind_payout if result > 0 else 0, "folds": 1 if play_bet == 0 else 0, "bad_folds": 1 if play_bet == 0 and player_rank < dealer_rank else 0 }

FUNCTIONrun_simulation(pre_flop_strategy, post_flop_strategy, river_strategy, dead_card_maker, hands):

    set total to 0

    set play_stats to { "player_wins": 0, "dealer_wins": 0, "pushes": 0, "dealer_qualifies": 0, "play_bet_total": 0, "player_win_play_bet_total": 0, "dealer_win_play_bet_total": 0, "push_play_bet_total": 0, "blind_payout_total": 0, "folds": 0, "bad_folds": 0 }

    do hands times:

        set result, round_data to play_game(pre_flop_strategy, post_flop_strategy, river_strategy, dead_card_maker)

        add result to total

        for each key in play_stats:

            add round_data.get(key, 0) to play_stats[key]

    print("Average result: {}".format(total / hands))

    for each key in play_stats:

        print("TOTAL {} : {}".format(key, play_stats[key]))

    print("________")

    for each key in play_stats:

        print("AVERAGE {} : {}".format(key, play_stats[key] / hands))

    return total / hands
