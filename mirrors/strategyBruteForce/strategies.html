FUNCTIONeval_even_if_unfull(board):

    set all_cards to hand + board

    if len(all_cards) < 5:

        set extras to []

        set ranks to sorted([Card.get_rank_int(card) for card in all_cards])

        // Check 1 card so we don't accidentally make a straight

        set new_suit to Card.next_suit(all_cards[0])

        // if highest card is less than 5 there's a risk we make a straight, so add a 6 to prevent

        if ranks[-1] < 4:

            extras.append(Card.new_from_ints(6, new_suit))

            ranks.append(6)

        while len(all_cards) + len(extras) < 5:

            for i from 0 to 5 - 1:

                if i not in ranks:

                    extras.append(Card.new_from_ints(i, new_suit))

                    ranks.append(i)

                    break

        set hand to hand+extras

    return evaluation.evaluate(hand, board)

FUNCTIONcount_outs(dead_cards):

    set player_score to evaluation.evaluate(player_hand, board)

    set used_cards to set(player_hand + board)

    set deck to Deck()

    deck.pull_many(used_cards)

    set remaining to deck.cards

    set outs to []

    for each card in remaining:

        set dealer_score to eval_even_if_unfull([card], board)

        if dealer_score < player_score:

            outs.append(card)

    return len(outs)

FUNCTIONbase_pre_flop(player_hand, dead_cards):

    if Card.get_rank_int(player_hand[0]) == 9 and Card.get_rank_int(player_hand[1]) >= 8:

        return True

    if Card.get_rank_int(player_hand[0]) >= 10 and Card.get_rank_int(player_hand[1]) == 6:

        return True

    if Card.get_rank_int(player_hand[0]) >= 11 and Card.get_rank_int(player_hand[1]) == 3:

        return True

    if Card.get_rank_int(player_hand[0]) >= 12:

        return True

FUNCTIONbase_post_flop(player_hand, board, dead_cards):

    set hand_rank to evaluation.evaluate(player_hand, board)

    if hand_rank <= 3321:

        return True

    // 5955 is highest pair of 3s rank

    // 6185 is highest pair rank

    if hand_rank <= 5955 and eval_even_if_unfull(board) > 6185:

        return True

    return False

FUNCTIONbase_river(player_hand, board, dead_cards):

    if count_outs(player_hand, board, dead_cards) < 22:

        return True

    return False
